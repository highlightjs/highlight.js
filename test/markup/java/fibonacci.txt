public class Fibonacci {
  public static void fibonacci(int n) {
    int a = 0, b = 1;
    for (int i = 0; i < n; i++) {
      System.out.print(a + " ");
      int sum = a + b;
      a = b;
      b = sum;
    }
  }
  public static void main(String[] args) {
    fibonacci(10); // Prints first 10 numbers
  }
}
